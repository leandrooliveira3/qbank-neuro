-- ========================================================
-- NOVA TABELA: MATERIAIS DIDÁTICOS (SYNC DRIVE)
-- ========================================================

CREATE TABLE IF NOT EXISTS public.didactic_materials (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  description TEXT,
  download_url TEXT,
  mime_type TEXT,
  size_bytes BIGINT,
  drive_path TEXT,
  drive_file_id TEXT UNIQUE, -- Garante unicidade para o Sync não duplicar
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Habilitar segurança (RLS)
ALTER TABLE public.didactic_materials ENABLE ROW LEVEL SECURITY;

-- Evita erro se a política já existir ao reexecutar o script
DROP POLICY IF EXISTS "Leitura pública para usuários autenticados" ON public.didactic_materials;

-- Política de Leitura: Todos os usuários autenticados podem ver e baixar os materiais
CREATE POLICY "Leitura pública para usuários autenticados" 
ON public.didactic_materials
FOR SELECT 
TO authenticated 
USING (true);
